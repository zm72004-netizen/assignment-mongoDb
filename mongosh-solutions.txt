// 1. Create an explicit collection named "books" with validation rule
db.createCollection("books", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["title"],
      properties: {
        title: {
          bsonType: "string",
          description: "must be a string and is required and cannot be empty"
        }
      }
    }
  }
});

// 2. Create an implicit collection by inserting data directly into "authors"
db.authors.insertOne({ "name": "Author1", "nationality": "British" });

// 3. Create a capped collection named "logs" with a size limit of 1MB (1048576 bytes)
db.createCollection("logs", { capped: true, size: 1048576 });

// 4. Create an index on the books collection for the title field
db.books.createIndex({ title: 1 });

// 5. Insert one document into the books collection
db.books.insertOne({
  "title": "Booki",
  "author": "Ali",
  "year": 1937,
  "genres": ["Fantasy", "Adventure"]
});

// 6. Insert multiple documents into the books collection
db.books.insertMany([
  {
    "title": "Future",
    "author": "George Orwell",
    "year": 2020,
    "genres": ["Science Fiction"]
  },
  {
    "title": "To Kill Mockingbird",
    "author": "Harper Lee",
    "year": 1960,
    "genres": ["Classic", "Fiction"]
  },
  {
    "title": "Brave New World",
    "author": "Aldous Huxley",
    "year": 2006,
    "genres": ["Dystopian", "Science Fiction"]
  }
]);

// 7. Insert a new log into the logs collection
db.logs.insertOne({
  "book_id": ObjectId("64b5c2d8a123456ef8914"),
  "action": "borrowed"
});

// 8. Update the book with title "Future" change the year to be 2022
db.books.updateOne(
  { title: "Future" },
  { $set: { year: 2022 } }
);

// 9. Find a Book with title "Brave New World"
db.books.findOne({ title: "Brave New World" });

// 10. Find all books published between 1990 and 2010
db.books.find({ year: { $gte: 1990, $lte: 2010 } });

// 11. Find books where the genre includes "Science Fiction"
db.books.find({ genres: "Science Fiction" });

// 12. Skip the first two books, limit the results to the next three, sorted by year in descending order
db.books.find().sort({ year: -1 }).skip(2).limit(3);

// 13. Find books where the year field stored as an integer
db.books.find({ year: { $type: "int" } });

// 14. Find all books where the genres field does not include "Horror" or "Science Fiction"
db.books.find({ genres: { $nin: ["Horror", "Science Fiction"] } });

// 15. Delete all books published before 2000
db.books.deleteMany({ year: { $lt: 2000 } });

// 16. Using aggregation: Filter books published after 2000 and sort them by year descending
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $sort: { year: -1 } }
]);

// 17. Using aggregation: Show only title, author, and year for books published after 2000
db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $project: { _id: 0, title: 1, author: 1, year: 1 } }
]);

// 18. Using aggregation: Break an array of genres into separate documents ($unwind)
db.books.aggregate([
  { $unwind: "$genres" },
  { $project: { title: 1, genres: 1 } }
]);

// 19. Using aggregation: Join the books collection with the logs collection ($lookup)
db.logs.aggregate([
  {
    $lookup: {
      from: "books",
      localField: "book_id",
      foreignField: "_id",
      as: "book_detail"
    }
  }
]);
